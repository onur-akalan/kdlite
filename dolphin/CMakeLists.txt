cmake_minimum_required(VERSION 3.16)
project(kdliteoverlay LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(KF6KIO REQUIRED)
add_library(kdliteoverlay MODULE kdliteoverlay.cpp)
set_target_properties(kdliteoverlay PROPERTIES PREFIX "")
target_link_libraries(kdliteoverlay PRIVATE Qt6::Core KF6::KIOCore)
add_executable(overlaytest EXCLUDE_FROM_ALL overlaytest.cpp)
target_link_libraries(overlaytest PRIVATE Qt6::Core KF6::KIOCore)
